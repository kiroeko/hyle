target_sources(hyle_engine PRIVATE main.cc weights.cc)

# Real-weight diagnostics remain explicit targets, separate from the application.
set(_qwen35_probes mtp_session)
if(APPLE AND "apple" IN_LIST _hyle_engine_routes)
  set(_qwen35_apple TRUE)
  set(_qwen35_native TRUE)
elseif("cuda" IN_LIST _hyle_engine_routes)
  set(_qwen35_cuda TRUE)
elseif("vulkan" IN_LIST _hyle_engine_routes)
  set(_qwen35_native TRUE)
endif()
if(_qwen35_native OR _qwen35_cuda)
  list(APPEND _qwen35_probes retirement)
endif()
if(_qwen35_native)
  list(APPEND _qwen35_probes retained_budget)
elseif(_qwen35_cuda)
  list(APPEND _qwen35_probes batch)
endif()
foreach(probe IN LISTS _qwen35_probes)
  add_executable(hyle_qwen35_${probe}_probe EXCLUDE_FROM_ALL ${probe}_probe.cc weights.cc)
  hyle_configure_engine_target(hyle_qwen35_${probe}_probe)
  if(probe STREQUAL "retirement")
    if(_qwen35_apple)
      enable_language(OBJCXX)
      target_sources(hyle_qwen35_retirement_probe PRIVATE native_gate_apple.mm)
      set_target_properties(hyle_qwen35_retirement_probe PROPERTIES
        OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
    else()
      target_sources(hyle_qwen35_retirement_probe PRIVATE native_gate.cc)
    endif()
  endif()
endforeach()
