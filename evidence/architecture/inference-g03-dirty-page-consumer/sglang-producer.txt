2813:         return self._get_value_buffer(layer_id)

2825:     def set_kv_buffer(
2826:         self,
2827:         layer: RadixAttention,
2828:         loc_info,
2829:         cache_k: torch.Tensor,
2830:         cache_v: torch.Tensor,
2831:         k_scale: Optional[float] = None,
2832:         v_scale: Optional[float] = None,
2833:         layer_id_override: Optional[int] = None,
2834:         dcp_kv_mask: Optional[torch.Tensor] = None,
2835:     ):
2836:         loc, _, _ = unwrap_write_loc(loc_info)
2837:         self._check_physical_write_loc(loc_info, "set_kv_buffer (MHA)")
2838:         # Catch stale slot ids here instead of as illegal-addr / silent KV
2839:         # corruption in the store_kvcache write (gated on SGLANG_ENABLE_ASYNC_ASSERT).
2840:         maybe_detect_oob(loc, 0, self.size + self.page_size, "set_kv_buffer (MHA)")
2841:         layer_id = (
2842:             layer_id_override if layer_id_override is not None else layer.layer_id
2843:         )
2844:         global_layer_id = layer.layer_id if layer is not None else layer_id
2845: 
2846:         if self.is_quantized_kv_cache:
2847:             if dcp_kv_mask is not None:
2848:                 raise RuntimeError("dcp_kv_mask is not supported for FP4 KV cache.")
2849:             self._set_quantized_kv_buffer(
2850:                 layer_id,
2851:                 global_layer_id,
2852:                 loc,
2853:                 cache_k,
2854:                 cache_v,
2855:                 k_scale,
2856:                 v_scale,
2857:             )
2858:             return
2859: 
2860:         if cache_k.dtype != self.dtype:
2861:             if k_scale is not None:
2862:                 cache_k.div_(k_scale)
2863:             if v_scale is not None:
2864:                 cache_v.div_(v_scale)
2865:             cache_k = cache_k.to(self.dtype)
2866:             cache_v = cache_v.to(self.dtype)
2867: 
2868:         if self.store_dtype != self.dtype:
2869:             cache_k = cache_k.view(self.store_dtype)
2870:             cache_v = cache_v.view(self.store_dtype)
2871: 
2872:         if dcp_kv_mask is not None:
2873:             N, H, D = cache_k.shape
2874:             k_buf = self.k_buffer[layer_id - self.start_layer]
2875:             v_buf = self.v_buffer[layer_id - self.start_layer]
2876:             masked_set_kv_buffer_kernel[(N,)](
2877:                 cache_k,
2878:                 cache_v,
2879:                 k_buf,
2880:                 v_buf,
2881:                 loc,
2882:                 dcp_kv_mask,
2883:                 N,
2884:                 H,
2885:                 D,
2886:                 128,
2887:                 cache_k.stride(0),
2888:                 cache_k.stride(1),
2889:                 cache_v.stride(0),
2890:                 cache_v.stride(1),
2891:                 k_buf.stride(0),
2892:                 v_buf.stride(0),
2893:             )
2894:             return
2895: 
2896:         if self.use_hnd:
2897:             # A slot is [page, :, off, :] (not a contiguous row), so scatter by (page, off).
2898:             k_buf = self.k_buffer[layer_id - self.start_layer]
2899:             v_buf = self.v_buffer[layer_id - self.start_layer]
2900:             if (
2901:                 _is_cuda
2902:                 and self.head_dim == self.v_head_dim
2903:                 and cache_k.stride()[1:] == cache_v.stride()[1:] == (self.head_dim, 1)
2904:             ):
2905:                 launch_reshape_and_cache_flash(
2906:                     cache_k,
2907:                     cache_v,
2908:                     k_buf.permute(0, 2, 1, 3),
2909:                     v_buf.permute(0, 2, 1, 3),
2910:                     loc,
2911:                 )
2912:                 return
2913:             pages = loc // self.page_size
2914:             offs = loc % self.page_size
2915:             k_buf[pages, :, offs, :] = cache_k
2916:             v_buf[pages, :, offs, :] = cache_v
2917:             return
2918: 
2919:         self._store_kv_layer(layer_id - self.start_layer, loc, cache_k, cache_v)

2923:         layer_idx: int,
2924:         loc: torch.Tensor,
2925:         cache_k: torch.Tensor,
2926:         cache_v: torch.Tensor,
2927:     ):

2929:         # layouts that change buffer identity (e.g. PageMajorMHATokenToKVPool's
2930:         # 4-D strided views). ``loc`` and the cache tensors are already dtype-cast

2932:         if self.kv_cache_layout == "vectorized_5d":

2938:             # The writer kernel uses key.stride(0) directly as the source
2939:             # token stride; head/dim are assumed contiguous within each
2940:             # token (stride(1)=head_size, stride(2)=1). Both hold for K/V
