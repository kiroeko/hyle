cmake_minimum_required(VERSION 3.25)
project(hyle_vulkan_gguf_delivery LANGUAGES CXX)

if(DEFINED HYLE_EMBED_SOURCE AND NOT HYLE_EMBED_SOURCE STREQUAL "")
  set(HYLE_COMPONENTS "compute;graph" CACHE STRING "" FORCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_CPU OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_CUDA OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_APPLE OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_ONEDNN OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_VULKAN ON CACHE BOOL "" FORCE)
  set(HYLE_STATIC OFF CACHE BOOL "" FORCE)
  add_subdirectory("${HYLE_EMBED_SOURCE}" hyle)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph)
endif()

add_executable(hyle_vulkan_gguf_delivery main.cc)
target_compile_features(hyle_vulkan_gguf_delivery PRIVATE cxx_std_23)
target_link_libraries(hyle_vulkan_gguf_delivery PRIVATE hyle::graph)
hyle_stage_runtime_dependencies(hyle_vulkan_gguf_delivery)

enable_testing()
add_test(NAME hyle_vulkan_gguf_delivery COMMAND hyle_vulkan_gguf_delivery)
