cmake_minimum_required(VERSION 3.25)
project(vulkan_v4_platform LANGUAGES CXX)
option(V4_ENABLE_VULKAN "Vulkan route" ON)
if(V4_SOURCE_ROOT)
  set(HYLE_COMPONENTS "compute,graph" CACHE STRING "" FORCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_TEST_INSTRUMENTATION OFF CACHE BOOL "" FORCE)
  set(HYLE_INSTALL ON CACHE BOOL "" FORCE)
  set(HYLE_STATIC ON CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_CPU OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPUTE_CUDA OFF CACHE STRING "" FORCE)
  set(HYLE_COMPUTE_APPLE OFF CACHE STRING "" FORCE)
  set(HYLE_COMPUTE_ONEDNN OFF CACHE STRING "" FORCE)
  if(V4_ENABLE_VULKAN)
    set(HYLE_COMPUTE_VULKAN ON CACHE STRING "" FORCE)
  else()
    set(HYLE_COMPUTE_VULKAN OFF CACHE STRING "" FORCE)
  endif()
  add_subdirectory("${V4_SOURCE_ROOT}" hyle)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute graph)
endif()
add_executable(v4_platform main.cc)
target_compile_features(v4_platform PRIVATE cxx_std_23)
target_link_libraries(v4_platform PRIVATE hyle::graph)
if(V4_ENABLE_VULKAN)
  target_compile_definitions(v4_platform PRIVATE V4_EXPECT_VULKAN=1)
endif()
