cmake_minimum_required(VERSION 3.28)
project(cpu_aten_numeric_probe LANGUAGES CXX)
find_package(Torch REQUIRED)
add_executable(cpu_aten_numeric_probe probe.cc)
target_compile_features(cpu_aten_numeric_probe PRIVATE cxx_std_20)
target_link_libraries(cpu_aten_numeric_probe PRIVATE torch torch_cpu c10)
if(TORCH_CXX_FLAGS)
  separate_arguments(flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
  target_compile_options(cpu_aten_numeric_probe PRIVATE ${flags})
endif()
