cmake_minimum_required(VERSION 3.25)
project(ConditionalAssetProducer LANGUAGES CXX)
include("C:/public/project-hyle/hyle/cmake/HyleRuntimeStaging.cmake")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/mlx.metallib"
  COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_BINARY_DIR}/mlx.metallib")
add_custom_target(asset_producer DEPENDS "${CMAKE_BINARY_DIR}/mlx.metallib")
add_library(asset_owner INTERFACE)
add_dependencies(asset_owner asset_producer)
set_property(TARGET asset_owner PROPERTY HYLE_APPLE_METALLIB "${CMAKE_BINARY_DIR}/mlx.metallib")
file(WRITE "${CMAKE_BINARY_DIR}/application.cc" "int main() { return 0; }\n")
add_executable(application "${CMAKE_BINARY_DIR}/application.cc")
set_property(TARGET application PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(application PRIVATE "$<$<BOOL:0>:asset_owner>")
hyle_stage_runtime(application)
