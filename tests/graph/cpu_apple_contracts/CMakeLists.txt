cmake_minimum_required(VERSION 3.25)
project(hyle_cpu_apple_contracts LANGUAGES CXX)
set(HYLE_COMPUTE_ROUTES "cpu;apple")
if(HYLE_CONTRACT_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS graph sched)
  add_subdirectory("${HYLE_CONTRACT_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph sched)
endif()
add_executable(hyle_cpu_apple_contracts ../cpu_apple_contracts.cc)
target_compile_features(hyle_cpu_apple_contracts PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_apple_contracts PRIVATE hyle::graph hyle::sched)
hyle_stage_runtime(hyle_cpu_apple_contracts)
option(HYLE_G09_NATIVE_TRACE "Build the original Metal observation fixture" OFF)
if(HYLE_G09_NATIVE_TRACE)
  add_executable(hyle_cpu_apple_pipeline ../cpu_apple_pipeline_trace.mm)
  # Scope the observation fixture's language to this source. Enabling OBJCXX
  # globally changes the language selection of the existing GGML .m sources.
  set_source_files_properties(../cpu_apple_pipeline_trace.mm PROPERTIES
    LANGUAGE CXX COMPILE_OPTIONS "-x;objective-c++")
  target_link_libraries(hyle_cpu_apple_pipeline PRIVATE "-framework Foundation" "-framework Metal")
else()
  add_executable(hyle_cpu_apple_pipeline ../native_joint_pipeline.cc)
endif()
target_compile_features(hyle_cpu_apple_pipeline PRIVATE cxx_std_23)
target_compile_definitions(hyle_cpu_apple_pipeline PRIVATE HYLE_G09_APPLE=1)
target_link_libraries(hyle_cpu_apple_pipeline PRIVATE hyle::graph hyle::sched)
hyle_stage_runtime(hyle_cpu_apple_pipeline)
enable_testing()
add_test(NAME hyle_cpu_apple_contracts COMMAND hyle_cpu_apple_contracts)
set_tests_properties(hyle_cpu_apple_contracts PROPERTIES TIMEOUT 180)
add_test(NAME hyle_cpu_apple_pipeline COMMAND hyle_cpu_apple_pipeline)
set_tests_properties(hyle_cpu_apple_pipeline PROPERTIES TIMEOUT 600)
