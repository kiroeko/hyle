cmake_minimum_required(VERSION 3.25)
project(hyle_cpu_cuda_contracts LANGUAGES CXX)
if(HYLE_CONTRACT_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS graph)
  set(HYLE_COMPUTE_ROUTES "cpu;cuda")
  add_subdirectory("${HYLE_CONTRACT_SOURCE}" hyle_build)
else()
  set(HYLE_COMPUTE_ROUTES "cpu;cuda")
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph)
endif()
add_executable(hyle_cpu_cuda_contracts ../cpu_cuda_contracts.cc)
target_compile_features(hyle_cpu_cuda_contracts PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_cuda_contracts PRIVATE hyle::graph)
hyle_stage_runtime(hyle_cpu_cuda_contracts)
enable_testing()
add_test(NAME hyle_cpu_cuda_contracts COMMAND hyle_cpu_cuda_contracts)
set_tests_properties(hyle_cpu_cuda_contracts PROPERTIES TIMEOUT 180)
