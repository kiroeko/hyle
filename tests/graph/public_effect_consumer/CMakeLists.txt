cmake_minimum_required(VERSION 3.25)
project(hyle_public_effect_consumer LANGUAGES CXX)
if(HYLE_CONTRACT_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS graph sched)
  add_subdirectory("${HYLE_CONTRACT_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph sched)
endif()
add_executable(hyle_public_effect_consumer ../public_effect_consumer.cc)
target_compile_features(hyle_public_effect_consumer PRIVATE cxx_std_23)
target_link_libraries(hyle_public_effect_consumer PRIVATE hyle::graph hyle::sched)
hyle_stage_runtime(hyle_public_effect_consumer)
enable_testing()
add_test(NAME hyle_public_effect_consumer COMMAND hyle_public_effect_consumer)
set_tests_properties(hyle_public_effect_consumer PROPERTIES TIMEOUT 180)
