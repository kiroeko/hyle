cmake_minimum_required(VERSION 3.25)
project(hyle_cpu_vulkan_contracts LANGUAGES CXX)
set(HYLE_COMPUTE_ROUTES "cpu;vulkan")
if(HYLE_CONTRACT_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS graph sched)
  add_subdirectory("${HYLE_CONTRACT_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph sched)
endif()
add_executable(hyle_cpu_vulkan_contracts ../cpu_vulkan_contracts.cc)
target_compile_features(hyle_cpu_vulkan_contracts PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_vulkan_contracts PRIVATE hyle::graph hyle::sched)
hyle_stage_runtime(hyle_cpu_vulkan_contracts)
add_executable(hyle_cpu_vulkan_pipeline ../native_joint_pipeline.cc)
target_compile_features(hyle_cpu_vulkan_pipeline PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_vulkan_pipeline PRIVATE hyle::graph hyle::sched)
hyle_stage_runtime(hyle_cpu_vulkan_pipeline)
option(HYLE_G09_NATIVE_TRACE "Build the original Vulkan math observation fixture" OFF)
if(HYLE_G09_NATIVE_TRACE)
  add_library(hyle_g09_vulkan_trace_loader SHARED ../vulkan_pipeline_trace_loader.cc)
  target_compile_features(hyle_g09_vulkan_trace_loader PRIVATE cxx_std_23)
  target_link_libraries(hyle_g09_vulkan_trace_loader PRIVATE hyle::graph ${CMAKE_DL_LIBS})
  add_executable(hyle_cpu_vulkan_pipeline_trace ../cpu_vulkan_pipeline_trace.cc)
  target_compile_features(hyle_cpu_vulkan_pipeline_trace PRIVATE cxx_std_23)
  target_link_libraries(hyle_cpu_vulkan_pipeline_trace PRIVATE hyle::graph hyle::sched)
  hyle_stage_runtime(hyle_cpu_vulkan_pipeline_trace)
endif()
enable_testing()
add_test(NAME hyle_cpu_vulkan_contracts COMMAND hyle_cpu_vulkan_contracts)
set_tests_properties(hyle_cpu_vulkan_contracts PROPERTIES TIMEOUT 180)
add_test(NAME hyle_cpu_vulkan_pipeline COMMAND hyle_cpu_vulkan_pipeline)
set_tests_properties(hyle_cpu_vulkan_pipeline PROPERTIES TIMEOUT 600)
