cmake_minimum_required(VERSION 3.25)
project(hyle_torch_sdk_cache_test LANGUAGES NONE)
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HYLE_SOURCE_DIR}/cmake/HyleRuntimeStaging.cmake")
include("${HYLE_SOURCE_DIR}/src/compute_provider_native/torch_cpu_fetch.cmake")

# Placeholder Windows SDK files exercise only configuration, never native code
# or downloads. Keep every generated fixture in this test's binary directory.
set(CMAKE_SYSTEM_NAME Windows)
set(WIN32 TRUE)
set(HYLE_TARGET_PROCESSOR x86_64)
set(CMAKE_FIND_LIBRARY_PREFIXES "")
set(CMAKE_FIND_LIBRARY_SUFFIXES ".lib")
unset(ENV{TORCH_INSTALL_PREFIX})
set(_sdk "${CMAKE_CURRENT_BINARY_DIR}/sdk")
file(MAKE_DIRECTORY "${_sdk}/lib" "${_sdk}/share/cmake/Torch")
file(WRITE "${_sdk}/build-version" "2.8.0+cpu\n")
foreach(_name IN ITEMS torch torch_cpu c10)
  file(WRITE "${_sdk}/lib/${_name}.lib" "fixture")
endforeach()
foreach(_name IN ITEMS asmjit c10 fbgemm libiomp5md torch torch_cpu uv)
  file(WRITE "${_sdk}/lib/${_name}.dll" "fixture")
endforeach()
file(WRITE "${_sdk}/share/cmake/Torch/TorchConfig.cmake" [=[
get_filename_component(TORCH_INSTALL_PREFIX "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
# The variable names and search calls match the locked TorchConfig's shared
# CPU branch, including optional kineto and its ordinary cache behavior.
find_library(TORCH_LIBRARY torch PATHS "${TORCH_INSTALL_PREFIX}/lib")
find_library(c10_LIBRARY c10 PATHS "${TORCH_INSTALL_PREFIX}/lib")
find_library(kineto_LIBRARY kineto PATHS "${TORCH_INSTALL_PREFIX}/lib")
if(NOT TARGET torch)
  foreach(_name IN ITEMS torch torch_cpu c10)
    add_library(${_name} UNKNOWN IMPORTED)
    set_property(TARGET ${_name} PROPERTY IMPORTED_LOCATION "${TORCH_INSTALL_PREFIX}/lib/${_name}.lib")
  endforeach()
  add_library(torch_library INTERFACE IMPORTED)
  add_library(torch_cpu_library INTERFACE IMPORTED)
endif()
set(CAFFE2_USE_CUDA OFF)
set(Torch_FOUND TRUE)
]=])
file(WRITE "${_sdk}/share/cmake/Torch/TorchConfigVersion.cmake" [=[
set(PACKAGE_VERSION 2.8.0)
set(PACKAGE_VERSION_COMPATIBLE TRUE)
set(PACKAGE_VERSION_EXACT TRUE)
]=])

set(HYLE_COMPUTE_TORCH_CPU_ROOT "${_sdk}")
foreach(_variable IN ITEMS TORCH_LIBRARY c10_LIBRARY kineto_LIBRARY Torch_DIR Caffe2_DIR)
  set(${_variable} "parent-${_variable}" CACHE STRING "parent-owned cache" FORCE)
endforeach()

# Reuse of the CUDA common targets must preserve the selected SDK identity,
# including configuration-specific locations and Windows import libraries.
include("${HYLE_SOURCE_DIR}/src/compute_provider_torch_cuda/torch_cuda_fetch.cmake")
# Installed shims in sibling directories may use separate copies of the same
# locked SDK. Their private imports must not collide through a global target.
set(_cuda_sibling "${CMAKE_CURRENT_BINARY_DIR}/cuda-sibling")
file(MAKE_DIRECTORY "${_cuda_sibling}")
file(WRITE "${_cuda_sibling}/CMakeLists.txt" [=[
set(_root "${CMAKE_CURRENT_BINARY_DIR}/sdk")
file(MAKE_DIRECTORY "${_root}/lib")
foreach(_name IN ITEMS c10_cuda torch_cuda)
  foreach(_suffix IN ITEMS dll lib)
    file(WRITE "${_root}/lib/${_name}.${_suffix}" "sibling fixture")
  endforeach()
endforeach()
hyle_import_torch_cuda_leaf("${_root}" "${_root}")
get_target_property(_global hyle_vendor_torch_cuda IMPORTED_GLOBAL)
get_target_property(_location hyle_vendor_torch_cuda IMPORTED_LOCATION)
if(_global OR NOT _location STREQUAL "${_root}/lib/torch_cuda.dll")
  message(FATAL_ERROR "A sibling CUDA shim reused another SDK's global target")
endif()
]=])
add_subdirectory("${_cuda_sibling}" "${CMAKE_CURRENT_BINARY_DIR}/cuda-left")
add_subdirectory("${_cuda_sibling}" "${CMAKE_CURRENT_BINARY_DIR}/cuda-right")
if(TARGET hyle_vendor_torch_cuda)
  message(FATAL_ERROR "A sibling CUDA shim leaked its private target")
endif()
set(_cuda_sdk "${CMAKE_CURRENT_BINARY_DIR}/cuda-sdk")
file(MAKE_DIRECTORY "${_cuda_sdk}/lib")
foreach(_name IN ITEMS c10_cuda torch_cuda)
  foreach(_suffix IN ITEMS dll lib)
    file(WRITE "${_cuda_sdk}/lib/${_name}.${_suffix}" "fixture")
  endforeach()
endforeach()
hyle_import_torch_cuda_leaf("${_cuda_sdk}" "${_sdk}")
if(HYLE_TORCH_TEST_CASE MATCHES "^cuda_foreign")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/other-cuda/lib")
  if(HYLE_TORCH_TEST_CASE STREQUAL "cuda_foreign_implib")
    set(_foreign_property IMPORTED_IMPLIB)
    set(_foreign_artifact "${CMAKE_CURRENT_BINARY_DIR}/other-cuda/lib/torch_cuda.lib")
  else()
    set(_foreign_property IMPORTED_LOCATION_RELEASE)
    set(_foreign_artifact "${CMAKE_CURRENT_BINARY_DIR}/other-cuda/lib/torch_cuda.dll")
    set_property(TARGET hyle_vendor_torch_cuda PROPERTY IMPORTED_CONFIGURATIONS RELEASE)
  endif()
  file(WRITE "${_foreign_artifact}" "foreign fixture")
  set_property(TARGET hyle_vendor_torch_cuda PROPERTY "${_foreign_property}" "${_foreign_artifact}")
endif()
hyle_import_torch_cuda_leaf("${_cuda_sdk}" "${_sdk}")

if(HYLE_TORCH_TEST_CASE STREQUAL "foreign")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/other-sdk/lib")
  foreach(_name IN ITEMS torch torch_cpu c10)
    set(_path "${CMAKE_CURRENT_BINARY_DIR}/other-sdk/lib/${_name}.lib")
    file(WRITE "${_path}" "foreign fixture")
    add_library(${_name} UNKNOWN IMPORTED)
    set_property(TARGET ${_name} PROPERTY IMPORTED_LOCATION "${_path}")
  endforeach()
  add_library(torch_library INTERFACE IMPORTED)
  add_library(torch_cpu_library INTERFACE IMPORTED)
elseif(HYLE_TORCH_TEST_CASE STREQUAL "partial")
  add_library(torch UNKNOWN IMPORTED)
  set_property(TARGET torch PROPERTY IMPORTED_LOCATION "${_sdk}/lib/torch.lib")
elseif(HYLE_TORCH_TEST_CASE STREQUAL "environment")
  set(ENV{TORCH_INSTALL_PREFIX} "${CMAKE_CURRENT_BINARY_DIR}/other-sdk")
endif()
_hyle_prepare_torch_cpu_sdk("${_sdk}")
find_package(Torch 2.8.0 EXACT CONFIG REQUIRED)
if(NOT TORCH_LIBRARY STREQUAL "${_sdk}/lib/torch.lib" OR
   NOT c10_LIBRARY STREQUAL "${_sdk}/lib/c10.lib" OR kineto_LIBRARY)
  message(FATAL_ERROR "The locked lookup consumed a foreign parent cache entry")
endif()

# Imported targets already owned by the parent must remain in that scope.
get_target_property(_before torch IMPORTED_GLOBAL)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/child")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/child/CMakeLists.txt"
  "_hyle_prepare_torch_cpu_sdk(\"${_sdk}\")\n_hyle_promote_torch_cpu_targets()\n"
  "hyle_import_torch_cuda_leaf(\"${_cuda_sdk}\" \"${_sdk}\" GLOBAL)\n")
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/child" "${CMAKE_CURRENT_BINARY_DIR}/child-build")
get_target_property(_after torch IMPORTED_GLOBAL)
if(NOT _before STREQUAL _after)
  message(FATAL_ERROR "A child changed a parent-owned imported target's scope")
endif()
get_target_property(_cuda_global hyle_vendor_torch_cuda IMPORTED_GLOBAL)
if(_cuda_global)
  message(FATAL_ERROR "A child promoted a parent-owned CUDA target to global scope")
endif()

# Exercise the real source entry and source/installed dependency shims too.
set(_HYLE_COMPUTE_REQUEST_CPU ON)
include("${HYLE_SOURCE_DIR}/src/compute_provider_native/torch_cpu_sdk.cmake")
set(_shim "${CMAKE_CURRENT_BINARY_DIR}/shim")
file(MAKE_DIRECTORY "${_shim}")
foreach(_file IN ITEMS torch_cpu_fetch.cmake HyleTorchCpuSdkConfig.cmake)
  configure_file("${HYLE_SOURCE_DIR}/src/compute_provider_native/${_file}" "${_shim}/${_file}" COPYONLY)
endforeach()
configure_file("${HYLE_SOURCE_DIR}/cmake/HyleDependencyLock.json" "${_shim}/HyleDependencyLock.json" COPYONLY)
configure_file("${HYLE_SOURCE_DIR}/src/compute_provider_torch_cuda/HyleTorchCudaSdkConfig.cmake"
  "${_shim}/HyleTorchCudaSdkConfig.cmake" COPYONLY)
file(WRITE "${_shim}/torch_cuda_fetch.cmake" [=[
function(hyle_resolve_torch_cuda_leaf lock_file output)
  set(${output} "unused-cuda-leaf-fixture" PARENT_SCOPE)
endfunction()
function(hyle_import_torch_cuda_leaf root cpu_root)
  if(NOT cpu_root STREQUAL HYLE_COMPUTE_TORCH_CPU_ROOT)
    message(FATAL_ERROR "CUDA shim did not reuse the selected CPU common SDK")
  endif()
endfunction()
]=])
include("${_shim}/HyleTorchCpuSdkConfig.cmake")
include("${_shim}/HyleTorchCudaSdkConfig.cmake")
foreach(_variable IN ITEMS TORCH_LIBRARY c10_LIBRARY kineto_LIBRARY Torch_DIR Caffe2_DIR)
  if(NOT "$CACHE{${_variable}}" STREQUAL "parent-${_variable}")
    message(FATAL_ERROR "A parent cache entry changed: ${_variable}")
  endif()
endforeach()

message(STATUS "Torch cache and matching-owner reuse checks passed")

enable_testing()
foreach(_case IN ITEMS foreign partial environment cuda_foreign cuda_foreign_implib)
  add_test(NAME torch_sdk_reject_${_case}
    COMMAND "${CMAKE_COMMAND}"
      "-DCASE=${_case}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
      "-DBINARY=${CMAKE_CURRENT_BINARY_DIR}/reject-${_case}"
      "-DGENERATOR=${CMAKE_GENERATOR}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/rejection.cmake")
endforeach()
