cmake_minimum_required(VERSION 3.25)
project(hyle_sanitizer_language_test LANGUAGES CXX CUDA ASM_MASM)
if(NOT MSVC)
  message(FATAL_ERROR "This fixture exercises MSVC, NVCC and MASM flag boundaries")
endif()
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(HYLE_SANITIZER asan)
set(HYLE_TARGET_PROCESSOR x86_64)
include("${HYLE_SOURCE_DIR}/cmake/HyleSanitizers.cmake")
add_library(mixed STATIC probe.cc probe.cu probe.asm)
hyle_apply_sanitizer(mixed)
foreach(_language IN ITEMS cxx cuda)
  add_executable(${_language}_consumer main.cc)
  if(_language STREQUAL "cuda")
    set_property(TARGET ${_language}_consumer PROPERTY LINKER_LANGUAGE CUDA)
  endif()
  target_link_libraries(${_language}_consumer PRIVATE mixed)
  hyle_apply_sanitizer(${_language}_consumer)
endforeach()
enable_testing()
add_test(NAME sanitizer_cxx_consumer COMMAND cxx_consumer)
add_test(NAME sanitizer_cuda_consumer COMMAND cuda_consumer)
