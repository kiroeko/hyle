cmake_minimum_required(VERSION 3.25)
project(hyle_state_consumer LANGUAGES CXX)
if(HYLE_FOUNDATION_SOURCE)
  set(HYLE_COMPONENTS "compute;graph" CACHE STRING "" FORCE)
  add_subdirectory("${HYLE_FOUNDATION_SOURCE}" hyle EXCLUDE_FROM_ALL)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute graph)
endif()
add_executable(hyle_state_consumer ../public_state_consumer.cc)
target_compile_features(hyle_state_consumer PRIVATE cxx_std_23)
target_link_libraries(hyle_state_consumer PRIVATE hyle::compute hyle::graph)
hyle_stage_runtime(hyle_state_consumer)
