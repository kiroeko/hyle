cmake_minimum_required(VERSION 3.25)
project(apple_parity_consumer LANGUAGES C CXX)

if(HYLE_SOURCE_ROOT)
  set(HYLE_COMPONENTS "compute,graph" CACHE STRING "" FORCE)
  set(HYLE_COMPUTE_ROUTES apple)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory("${HYLE_SOURCE_ROOT}" hyle)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute graph)
endif()

add_executable(apple_parity_consumer main.cc)
target_compile_features(apple_parity_consumer PRIVATE cxx_std_23)
target_link_libraries(apple_parity_consumer PRIVATE hyle::graph)
hyle_stage_apple_runtime(apple_parity_consumer)
enable_testing()
add_test(NAME ordinary_apple_delivery COMMAND apple_parity_consumer)
