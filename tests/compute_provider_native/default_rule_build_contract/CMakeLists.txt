cmake_minimum_required(VERSION 3.25)
project(HyleNativeRuleBuildContract NONE)
add_library(hyle_compute_provider_native INTERFACE)
include("${CMAKE_CURRENT_LIST_DIR}/../../../src/compute_provider_native/default_rule_build.cmake")

# This checks build-metadata projection only. Actual compiler/CPU execution is
# checked by the ordinary product consumers, not emulated by these variables.
set(CMAKE_SYSTEM_NAME Windows)
set(HYLE_TARGET_PROCESSOR x86_64)
set(CMAKE_CXX_COMPILER_ID MSVC)
set(CMAKE_CXX_COMPILER_VERSION 19.50.35728)
set(CMAKE_CXX_FLAGS "/DWIN32 /D_WINDOWS /EHsc")
set(CMAKE_CXX_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
set(HYLE_STATIC ON)
set(HYLE_COMPUTE_ONEDNN_ENABLED ON)
set(HYLE_SANITIZER off)
set(HYLE_SIMD_EXACT_TARGET "")
set(HYLE_TRACE OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)

function(expect_recipe name expected)
  hyle_native_default_rule_recipe(actual)
  if(NOT "${actual}" STREQUAL "${expected}")
    message(FATAL_ERROR "${name}: recipe ${actual}, expected ${expected}")
  endif()
  message(STATUS "${name}: ${actual}")
endfunction()
expect_recipe(matching_static TRUE)

foreach(_case IN ITEMS flags release_flags ipo release_ipo shared absent_peer
    sanitizer exact_isa trace compiler compiler_version platform processor runtime)
  block()
    if(_case STREQUAL "flags")
      string(APPEND CMAKE_CXX_FLAGS " /fp:fast")
    elseif(_case STREQUAL "release_flags")
      set(CMAKE_CXX_FLAGS_RELEASE "/Od /DNDEBUG")
    elseif(_case STREQUAL "ipo")
      set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
    elseif(_case STREQUAL "release_ipo")
      set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    elseif(_case STREQUAL "shared")
      set(HYLE_STATIC OFF)
    elseif(_case STREQUAL "absent_peer")
      set(HYLE_COMPUTE_ONEDNN_ENABLED OFF)
    elseif(_case STREQUAL "sanitizer")
      set(HYLE_SANITIZER asan)
    elseif(_case STREQUAL "exact_isa")
      set(HYLE_SIMD_EXACT_TARGET AVX2)
    elseif(_case STREQUAL "trace")
      set(HYLE_TRACE ON)
    elseif(_case STREQUAL "compiler")
      set(CMAKE_CXX_COMPILER_ID Clang)
    elseif(_case STREQUAL "compiler_version")
      set(CMAKE_CXX_COMPILER_VERSION 19.50.35729)
    elseif(_case STREQUAL "platform")
      set(CMAKE_SYSTEM_NAME Darwin)
    elseif(_case STREQUAL "processor")
      set(HYLE_TARGET_PROCESSOR arm64)
    elseif(_case STREQUAL "runtime")
      set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
    expect_recipe(${_case} FALSE)
  endblock()
endforeach()

set_property(TARGET hyle_compute_provider_native PROPERTY INTERPROCEDURAL_OPTIMIZATION ON)
expect_recipe(target_ipo FALSE)
set_property(TARGET hyle_compute_provider_native PROPERTY INTERPROCEDURAL_OPTIMIZATION OFF)
set_property(TARGET hyle_compute_provider_native PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
expect_recipe(target_release_ipo FALSE)
set_property(TARGET hyle_compute_provider_native PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
add_compile_options(/Od)
expect_recipe(parent_options FALSE)
set_property(DIRECTORY PROPERTY COMPILE_OPTIONS "")
add_compile_definitions(NDEBUG)
expect_recipe(parent_definitions FALSE)
set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS "")
expect_recipe(restored_static TRUE)
