cmake_minimum_required(VERSION 3.25)
project(hyle_cpu_vulkan_sample LANGUAGES CXX)
if(HYLE_SAMPLE_SOURCE)
  if(NOT DEFINED HYLE_COMPUTE_ROUTES)
    set(HYLE_COMPUTE_ROUTES "cpu;vulkan")
  endif()
  set(HYLE_COMPONENTS graph)
  add_subdirectory("${HYLE_SAMPLE_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph)
endif()
get_target_property(_sample_routes hyle::graph HYLE_COMPUTE_ROUTES)
foreach(_required_route IN ITEMS cpu vulkan)
  if(NOT _required_route IN_LIST _sample_routes)
    message(FATAL_ERROR "This sample requires the ${_required_route} route")
  endif()
endforeach()
unset(_sample_routes)
unset(_required_route)
add_executable(hyle_cpu_vulkan_sample main.cc)
target_compile_features(hyle_cpu_vulkan_sample PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_vulkan_sample PRIVATE hyle::graph)
hyle_stage_runtime(hyle_cpu_vulkan_sample)
enable_testing()
add_test(NAME hyle_cpu_vulkan_sample COMMAND hyle_cpu_vulkan_sample)
set_tests_properties(hyle_cpu_vulkan_sample PROPERTIES TIMEOUT 180)
