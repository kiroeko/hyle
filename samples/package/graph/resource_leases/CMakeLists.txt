cmake_minimum_required(VERSION 3.25)
project(hyle_resource_leases_sample LANGUAGES CXX)

if(HYLE_SAMPLE_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS graph)
  add_subdirectory("${HYLE_SAMPLE_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph)
endif()

add_executable(hyle_resource_leases_sample main.cc)
target_compile_features(hyle_resource_leases_sample PRIVATE cxx_std_23)
target_link_libraries(hyle_resource_leases_sample PRIVATE hyle::graph)
hyle_stage_runtime(hyle_resource_leases_sample)
enable_testing()
add_test(NAME hyle_resource_leases_sample COMMAND hyle_resource_leases_sample)
set_tests_properties(hyle_resource_leases_sample PROPERTIES TIMEOUT 180)
