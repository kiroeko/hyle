cmake_minimum_required(VERSION 3.25)
project(hyle_cpu_executor_sample LANGUAGES CXX)

option(HYLE_SAMPLE_ONEDNN "Also exercise the oneDNN CPU region" OFF)
if(HYLE_SAMPLE_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS "graph;sched")
  set(HYLE_COMPUTE_CUDA OFF CACHE STRING "" FORCE)
  if(HYLE_SAMPLE_ONEDNN)
    set(HYLE_COMPUTE_ONEDNN ON CACHE STRING "" FORCE)
  else()
    set(HYLE_COMPUTE_ONEDNN OFF CACHE STRING "" FORCE)
  endif()
  add_subdirectory("${HYLE_SAMPLE_SOURCE}" hyle_build)
  if(NOT HYLE_COMPUTE_CPU)
    message(FATAL_ERROR "The CPU executor sample requires HYLE_COMPUTE_CPU=ON")
  endif()
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS graph sched)
  if(NOT "compute_cpu" IN_LIST Hyle_AVAILABLE_FEATURES)
    message(FATAL_ERROR "The CPU executor sample requires an SDK with compute_cpu")
  endif()
  if(HYLE_SAMPLE_ONEDNN AND NOT "compute_onednn" IN_LIST Hyle_AVAILABLE_FEATURES)
    message(FATAL_ERROR "This SDK does not provide the requested oneDNN route")
  endif()
endif()

add_executable(hyle_cpu_executor_sample main.cc)
target_compile_features(hyle_cpu_executor_sample PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_executor_sample PRIVATE hyle::graph hyle::sched)
if(HYLE_SAMPLE_ONEDNN)
  target_compile_definitions(hyle_cpu_executor_sample PRIVATE HYLE_SAMPLE_ONEDNN=1)
endif()
hyle_stage_runtime_dependencies(hyle_cpu_executor_sample)
enable_testing()
add_test(NAME hyle_cpu_executor_sample COMMAND hyle_cpu_executor_sample)
set_tests_properties(hyle_cpu_executor_sample PROPERTIES TIMEOUT 120)
