cmake_minimum_required(VERSION 3.25)
project(hyle_apple_weights_sample LANGUAGES CXX)
set(HYLE_COMPUTE_ROUTES apple)
if(HYLE_SAMPLE_SOURCE)
  set(HYLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HYLE_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(HYLE_COMPONENTS compute)
  add_subdirectory("${HYLE_SAMPLE_SOURCE}" hyle_build)
else()
  find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute)
endif()
add_executable(hyle_apple_weights_sample main.cc)
target_compile_features(hyle_apple_weights_sample PRIVATE cxx_std_23)
target_link_libraries(hyle_apple_weights_sample PRIVATE hyle::compute)
hyle_stage_runtime(hyle_apple_weights_sample)
enable_testing()
add_test(NAME hyle_apple_weights_sample COMMAND hyle_apple_weights_sample)
set_tests_properties(hyle_apple_weights_sample PROPERTIES TIMEOUT 120)
