cmake_minimum_required(VERSION 3.25)
project(g07_packed_weights_p1 LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(HYLE_G07_ROUTE "" CACHE STRING "cuda or apple")
set(HYLE_G07_REFERENCE_SOURCE "" CACHE PATH "Clean original product669 checkout")
if(NOT HYLE_G07_ROUTE MATCHES "^(cuda|apple)$")
  message(FATAL_ERROR "Specify HYLE_G07_ROUTE=cuda or apple")
endif()
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Hyle CONFIG REQUIRED COMPONENTS compute)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT Hyle_STATIC)
  message(FATAL_ERROR "Use the independently qualified static SDK")
endif()
set(_original "${CMAKE_CURRENT_BINARY_DIR}/original669")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_reference.py"
  --original "${HYLE_G07_REFERENCE_SOURCE}" --output "${_original}" --route "${HYLE_G07_ROUTE}"
  RESULT_VARIABLE _prepared OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/source-generation.log")
if(NOT _prepared EQUAL 0)
  message(FATAL_ERROR "Original669 source identity/materialization failed")
endif()
add_library(g07_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g07_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
if(HYLE_G07_ROUTE STREQUAL "cuda")
  set(_peer "${_original}/artifact_provider.cc")
else()
  if(NOT APPLE)
    message(FATAL_ERROR "Apple requires the actual MacBook Air")
  endif()
  set(_peer "${_original}/apple_mature_float.cc")
endif()
foreach(_variant IN ITEMS primary placement)
  set(_target g07_packed_${_variant})
  if(_variant STREQUAL "placement")
    add_executable(${_target} placement.cc packed_bench.cc "${_peer}")
    target_compile_definitions(${_target} PRIVATE HYLE_G07_PLACEMENT=1)
  else()
    add_executable(${_target} packed_bench.cc "${_peer}")
    target_compile_definitions(${_target} PRIVATE HYLE_G07_PLACEMENT=0)
  endif()
  target_compile_features(${_target} PRIVATE cxx_std_23)
  target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(${_target} PRIVATE "${_original}" "${HYLE_SOURCE_DIR}/bench/common")
  target_link_libraries(${_target} PRIVATE hyle::compute g07_nanobench)
  if(HYLE_G07_ROUTE STREQUAL "cuda")
    target_compile_definitions(${_target} PRIVATE HYLE_G07_CUDA=1)
    target_include_directories(${_target} PRIVATE
      "${HYLE_SOURCE_DIR}/src/compute" "${HYLE_SOURCE_DIR}/src/compute_backend_cuda"
      "${HYLE_SOURCE_DIR}/third_party/bitsandbytes_cuda" "${HYLE_SOURCE_DIR}/third_party/awq_cuda"
      "${HYLE_SOURCE_DIR}/third_party/compressed_tensors_cuda")
    target_include_directories(${_target} SYSTEM PRIVATE
      "${_hyle_cpu_root}/include" "${_hyle_cpu_root}/include/torch/csrc/api/include" "${_hyle_cuda_root}/include")
    target_link_libraries(${_target} PRIVATE hyle::_factor_compute_provider_torch_cuda
      torch torch_cpu c10 hyle_vendor_torch_cuda hyle_vendor_c10_cuda CUDA::cudart CUDA::cublas)
    if(TORCH_CXX_FLAGS)
      separate_arguments(_torch_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
      target_compile_options(${_target} PRIVATE ${_torch_flags})
    endif()
    target_link_options(${_target} PRIVATE LINKER:--disable-new-dtags)
  else()
    target_include_directories(${_target} PRIVATE "${HYLE_SOURCE_DIR}/src/compute_route_apple")
  endif()
  set_property(TARGET ${_target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  hyle_stage_runtime(${_target})
endforeach()
add_executable(g07_packed_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g07_packed_control PRIVATE cxx_std_23)
target_link_libraries(g07_packed_control PRIVATE hyle::compute g07_nanobench)
set_property(TARGET g07_packed_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g07_packed_control)
