cmake_minimum_required(VERSION 3.25)
project(g04_vulkan_encoded_p1 LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This finite contract is bound to the actual MacBook Air")
endif()
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(HYLE_RESIDENT_REFERENCE_SOURCE "" CACHE PATH "Authenticated original ggml peer")
set(HYLE_RESIDENT_GLSLANG "" CACHE FILEPATH "Qualified GLSL compiler")
find_package(Hyle CONFIG REQUIRED COMPONENTS compute graph)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT Hyle_STATIC)
  message(FATAL_ERROR "Original private recipe adapter needs the qualified static SDK")
endif()
set(_source "${HYLE_SOURCE_DIR}/bench/inference_resident_sdpa")
set(_generated "${CMAKE_CURRENT_BINARY_DIR}/reference-shaders")
add_custom_command(OUTPUT "${_generated}/reference_spirv.h"
  COMMAND "${Python3_EXECUTABLE}" "${_source}/reference_shaders.py"
    --peer "${HYLE_RESIDENT_REFERENCE_SOURCE}" --compiler "${HYLE_RESIDENT_GLSLANG}"
    --output "${_generated}"
  DEPENDS "${_source}/reference_shaders.py" VERBATIM)
add_library(g04_encoded_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g04_encoded_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
foreach(variant primary placement)
  if(variant STREQUAL "primary")
    set(target g04_vulkan_encoded_bench)
  else()
    set(target g04_vulkan_encoded_bench_placement)
  endif()
  add_executable(${target} "${_source}/vulkan_encoded_page_bench.cc"
    "${_source}/vulkan_encoded_page_reference.cc" "${_source}/reference.cc"
    "${_generated}/reference_spirv.h")
  if(variant STREQUAL "placement")
    target_sources(${target} PRIVATE "${_source}/placement.cc")
    target_compile_definitions(${target} PRIVATE HYLE_ENCODED_PAGE_PLACEMENT=1)
  else()
    target_compile_definitions(${target} PRIVATE HYLE_ENCODED_PAGE_PLACEMENT=0)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(${target} PRIVATE "${_generated}"
    "${HYLE_SOURCE_DIR}/src/compute_route_vulkan" "${HYLE_SOURCE_DIR}/bench/common")
  target_link_libraries(${target} PRIVATE hyle::compute g04_encoded_nanobench)
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  hyle_stage_runtime(${target})
endforeach()
add_executable(g04_vulkan_encoded_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g04_vulkan_encoded_control PRIVATE cxx_std_23)
target_link_libraries(g04_vulkan_encoded_control PRIVATE hyle::compute g04_encoded_nanobench)
set_property(TARGET g04_vulkan_encoded_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g04_vulkan_encoded_control)
