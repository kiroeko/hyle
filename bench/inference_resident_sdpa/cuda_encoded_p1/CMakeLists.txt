cmake_minimum_required(VERSION 3.25)
project(g04_cuda_encoded_p1 LANGUAGES C CXX CUDA)
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Hyle CONFIG REQUIRED COMPONENTS compute graph)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cuda_encoded_peer" original-peer)
add_executable(g04_cuda_encoded_bench_placement "../cpu_placement.cc"
  "../cuda_encoded_page_bench.cc" "../cuda_encoded_page_reference.cc")
target_compile_features(g04_cuda_encoded_bench_placement PRIVATE cxx_std_23)
target_compile_definitions(g04_cuda_encoded_bench_placement PRIVATE HYLE_RESIDENT_PLACEMENT=1)
target_compile_options(g04_cuda_encoded_bench_placement PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(g04_cuda_encoded_bench_placement PRIVATE "${HYLE_SOURCE_DIR}/src/compute_backend_cuda")
target_link_libraries(g04_cuda_encoded_bench_placement PRIVATE hyle::compute g04_cuda_encoded_original g04_cuda_encoded_nanobench)
set_property(TARGET g04_cuda_encoded_bench_placement PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g04_cuda_encoded_bench_placement)
add_executable(g04_cuda_encoded_perf_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g04_cuda_encoded_perf_control PRIVATE cxx_std_23)
target_link_libraries(g04_cuda_encoded_perf_control PRIVATE hyle::compute g04_cuda_encoded_nanobench)
set_property(TARGET g04_cuda_encoded_perf_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g04_cuda_encoded_perf_control)
