cmake_minimum_required(VERSION 3.25)
project(g04_cuda_encoded_peer LANGUAGES C CXX CUDA)
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Hyle CONFIG REQUIRED COMPONENTS compute graph)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT Hyle_STATIC OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Use the qualified Linux CUDA static SDK")
endif()
set(HYLE_FLASHINFER_ORIGINAL_SOURCE "" CACHE PATH "Pristine authenticated FlashInfer69ff checkout")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_source.py"
  --original "${HYLE_FLASHINFER_ORIGINAL_SOURCE}"
  --prepared "${CMAKE_CURRENT_BINARY_DIR}/sync-fixed-flashinfer"
  --output "${CMAKE_CURRENT_BINARY_DIR}/original-flashinfer-source.json"
  RESULT_VARIABLE verified)
if(NOT verified EQUAL 0)
  message(FATAL_ERROR "Original FlashInfer identity check failed")
endif()
set(_source "${HYLE_SOURCE_DIR}/bench/inference_resident_sdpa")
add_library(g04_cuda_encoded_original STATIC "${_source}/cuda_encoded_page_native.cu")
set_property(TARGET g04_cuda_encoded_original PROPERTY CUDA_ARCHITECTURES 89-real)
set_property(TARGET g04_cuda_encoded_original PROPERTY CUDA_STANDARD 20)
set_property(TARGET g04_cuda_encoded_original PROPERTY JOB_POOL_COMPILE g04_cuda_peer_compile)
set_property(GLOBAL APPEND PROPERTY JOB_POOLS g04_cuda_peer_compile=1)
target_include_directories(g04_cuda_encoded_original PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/sync-fixed-flashinfer/include")
target_compile_options(g04_cuda_encoded_original PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
add_library(g04_cuda_encoded_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g04_cuda_encoded_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
add_executable(g04_cuda_encoded_bench "${_source}/cuda_encoded_page_bench.cc"
  "${_source}/cuda_encoded_page_reference.cc")
target_compile_features(g04_cuda_encoded_bench PRIVATE cxx_std_23)
target_compile_options(g04_cuda_encoded_bench PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(g04_cuda_encoded_bench PRIVATE "${HYLE_SOURCE_DIR}/src/compute_backend_cuda")
target_link_libraries(g04_cuda_encoded_bench PRIVATE hyle::compute g04_cuda_encoded_original g04_cuda_encoded_nanobench)
set_property(TARGET g04_cuda_encoded_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g04_cuda_encoded_bench)
