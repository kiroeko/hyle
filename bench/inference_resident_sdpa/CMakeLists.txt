# Narrow adapter for the G01 immutable candidate; no changes to the public
# performance runner or HYLE's component/build graph.
cmake_minimum_required(VERSION 3.25)
if(NOT TARGET hyle_graph_route_vulkan)
  project(g01_resident_sdpa_p1 LANGUAGES C CXX)
  get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  set(HYLE_COMPONENTS "compute;graph")
  set(HYLE_COMPUTE_ROUTES vulkan)
  set(HYLE_BUILD_TESTS OFF)
  set(HYLE_BUILD_BENCH OFF)
  add_subdirectory("${HYLE_SOURCE_DIR}" hyle)
endif()
if(NOT HYLE_STATIC)
  message(FATAL_ERROR "Resident native reference needs a static source build for private lifecycle adaptation")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(_resident_bench_source "${CMAKE_CURRENT_LIST_DIR}")
set(_resident_reference "${CMAKE_CURRENT_BINARY_DIR}/reference-shaders")
if(NOT EXISTS "${HYLE_RESIDENT_REFERENCE_SOURCE}/src/ggml-vulkan/ggml-vulkan.cpp")
  message(FATAL_ERROR "HYLE_RESIDENT_REFERENCE_SOURCE must name the pinned original ggml checkout")
endif()
add_custom_command(OUTPUT "${_resident_reference}/reference_spirv.h"
  COMMAND "${Python3_EXECUTABLE}" "${_resident_bench_source}/reference_shaders.py"
    --peer "${HYLE_RESIDENT_REFERENCE_SOURCE}" --compiler "${HYLE_RESIDENT_GLSLANG}"
    --output "${_resident_reference}"
  DEPENDS "${_resident_bench_source}/reference_shaders.py" VERBATIM)
add_library(g01_resident_reference STATIC "${_resident_bench_source}/reference.cc" "${_resident_reference}/reference_spirv.h")
target_include_directories(g01_resident_reference PRIVATE "${_resident_reference}"
  "${HYLE_SOURCE_DIR}/src/compute_route_vulkan")
target_link_libraries(g01_resident_reference PRIVATE hyle_compute_route_vulkan)
target_compile_features(g01_resident_reference PRIVATE cxx_std_23)
hyle_apply_warnings(g01_resident_reference)
add_library(g01_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g01_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
add_executable(g01_resident_bench "${_resident_bench_source}/route_bench.cc")
target_compile_features(g01_resident_bench PRIVATE cxx_std_23)
target_link_libraries(g01_resident_bench PRIVATE hyle::graph g01_resident_reference g01_nanobench)
hyle_apply_warnings(g01_resident_bench)
if(MSVC)
  target_compile_definitions(g01_resident_bench PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(APPLE)
  find_file(_resident_loader NAMES libvulkan.1.dylib HINTS /opt/homebrew/opt/vulkan-loader/lib REQUIRED)
  target_compile_definitions(g01_resident_bench PRIVATE HYLE_RESIDENT_VULKAN_LOADER="${_resident_loader}")
endif()
set_property(TARGET g01_resident_bench g01_resident_reference PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g01_resident_bench)
add_executable(g01_perf_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g01_perf_control PRIVATE cxx_std_23)
target_link_libraries(g01_perf_control PRIVATE hyle::graph g01_nanobench)
set_property(TARGET g01_perf_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g01_perf_control)
