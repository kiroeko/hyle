cmake_minimum_required(VERSION 3.25)
if(NOT TARGET hyle_compute_route_cpu)
  project(g01_cpu_resident_p1 LANGUAGES C CXX)
  get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  set(HYLE_COMPONENTS "compute;graph")
  set(HYLE_COMPUTE_ROUTES cpu)
  set(HYLE_BUILD_TESTS OFF)
  set(HYLE_BUILD_BENCH OFF)
  add_subdirectory("${HYLE_SOURCE_DIR}" hyle)
endif()
if(NOT HYLE_STATIC)
  message(FATAL_ERROR "CPU original native reference requires static source linkage")
endif()
set(_cpu_bench_source "${CMAKE_CURRENT_LIST_DIR}")
add_library(g01_cpu_reference STATIC "${_cpu_bench_source}/cpu_reference.cc")
target_compile_features(g01_cpu_reference PRIVATE cxx_std_23)
target_include_directories(g01_cpu_reference PRIVATE "${HYLE_SOURCE_DIR}/src/compute_ggml_runtime")
target_link_libraries(g01_cpu_reference PRIVATE hyle::compute hyle_compute_ggml_runtime ggml)
hyle_apply_warnings(g01_cpu_reference)
add_library(g01_cpu_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g01_cpu_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
add_executable(g01_cpu_resident_bench "${_cpu_bench_source}/cpu_bench.cc")
target_compile_features(g01_cpu_resident_bench PRIVATE cxx_std_23)
target_link_libraries(g01_cpu_resident_bench PRIVATE hyle::compute g01_cpu_reference g01_cpu_nanobench)
hyle_apply_warnings(g01_cpu_resident_bench)
if(MSVC)
  target_compile_definitions(g01_cpu_resident_bench PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
set_property(TARGET g01_cpu_resident_bench g01_cpu_reference PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g01_cpu_resident_bench)
