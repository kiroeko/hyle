cmake_minimum_required(VERSION 3.25)
project(g01_cpu_paged_p1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This CPU reference consumer uses Windows DLL loading and MSVC-compatible options")
endif()
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(HYLE_CPU_ORIGINAL_REFERENCE_SOURCE "" CACHE PATH "Pristine ggml source pin")
set(HYLE_CPU_PAGED_ORIGINAL_DLL "" CACHE FILEPATH "Independently built pristine CPU artifact")
set(HYLE_CPU_PAGED_NATIVE_INCLUDE "" CACHE PATH "Candidate native headers matching the SDK")
if(NOT EXISTS "${HYLE_CPU_PAGED_NATIVE_INCLUDE}/ggml.h" OR
   NOT EXISTS "${HYLE_CPU_PAGED_NATIVE_INCLUDE}/ggml-cpu.h")
  message(FATAL_ERROR "HYLE_CPU_PAGED_NATIVE_INCLUDE must contain the ggml.h and ggml-cpu.h matching the SDK's native runtime")
endif()
if(NOT HYLE_CPU_PAGED_ORIGINAL_DLL OR NOT EXISTS "${HYLE_CPU_PAGED_ORIGINAL_DLL}")
  message(FATAL_ERROR "HYLE_CPU_PAGED_ORIGINAL_DLL must name the independently built original CPU reference DLL")
endif()
set(HYLE_COMPUTE_ROUTES cpu)
find_package(Hyle CONFIG REQUIRED COMPONENTS compute graph)
if(NOT Hyle_STATIC OR NOT TARGET hyle::_factor_compute_ggml_runtime)
  message(FATAL_ERROR "The CPU reference consumer requires a static Hyle SDK with the CPU route")
endif()
add_library(g01_paged_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g01_paged_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
foreach(variant primary placement)
  if(variant STREQUAL "primary")
    set(target g01_cpu_paged_bench)
  else()
    set(target g01_cpu_paged_bench_placement)
  endif()
  add_executable(${target} "../cpu_paged_bench.cc" "../cpu_paged_reference.cc")
  if(variant STREQUAL "placement")
    target_sources(${target} PRIVATE "../cpu_placement.cc")
    target_compile_definitions(${target} PRIVATE HYLE_RESIDENT_PLACEMENT=1)
  else()
    target_compile_definitions(${target} PRIVATE HYLE_RESIDENT_PLACEMENT=0)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_include_directories(${target} PRIVATE "${HYLE_SOURCE_DIR}/src/compute_ggml_runtime" "${HYLE_CPU_PAGED_NATIVE_INCLUDE}")
  target_compile_definitions(${target} PRIVATE HYLE_CPU_PAGED_ORIGINAL_DLL="${HYLE_CPU_PAGED_ORIGINAL_DLL}")
  target_compile_options(${target} PRIVATE /W4 /WX /wd4324 /utf-8)
  target_link_libraries(${target} PRIVATE hyle::compute hyle::_factor_compute_ggml_runtime g01_paged_nanobench)
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  hyle_stage_runtime(${target})
endforeach()
add_executable(g01_cpu_paged_perf_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g01_cpu_paged_perf_control PRIVATE cxx_std_23)
target_link_libraries(g01_cpu_paged_perf_control PRIVATE hyle::compute g01_paged_nanobench)
set_property(TARGET g01_cpu_paged_perf_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g01_cpu_paged_perf_control)
