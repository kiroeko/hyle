cmake_minimum_required(VERSION 3.25)
project(g01_cpu_ragged_p1 LANGUAGES C CXX)
get_filename_component(HYLE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(HYLE_CPU_ORIGINAL_REFERENCE_SOURCE "" CACHE PATH "Authenticated original CPU ggml peer checkout")
set(HYLE_COMPONENTS "compute;graph")
set(HYLE_COMPUTE_ROUTES cpu)
set(HYLE_BUILD_TESTS OFF)
set(HYLE_BUILD_BENCH OFF)
add_subdirectory("${HYLE_SOURCE_DIR}" hyle)
if(NOT HYLE_STATIC)
  message(FATAL_ERROR "Original CPU ragged numerical comparator uses static source linkage")
endif()
add_library(g01_ragged_nanobench STATIC "${HYLE_SOURCE_DIR}/bench/nanobench_impl.cc")
target_include_directories(g01_ragged_nanobench SYSTEM PUBLIC "${HYLE_SOURCE_DIR}/third_party/nanobench")
foreach(variant primary placement)
  if(variant STREQUAL "primary")
    set(target g01_cpu_ragged_bench)
  else()
    set(target g01_cpu_ragged_bench_placement)
  endif()
  add_executable(${target} "../cpu_ragged_bench.cc" "../cpu_ragged_reference.cc")
  if(variant STREQUAL "placement")
    target_sources(${target} PRIVATE "../cpu_placement.cc")
    target_compile_definitions(${target} PRIVATE HYLE_RESIDENT_PLACEMENT=1)
  else()
    target_compile_definitions(${target} PRIVATE HYLE_RESIDENT_PLACEMENT=0)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_include_directories(${target} PRIVATE "${HYLE_SOURCE_DIR}/src/compute_ggml_runtime")
  target_link_libraries(${target} PRIVATE hyle::compute hyle_compute_ggml_runtime ggml g01_ragged_nanobench)
  hyle_apply_warnings(${target})
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  hyle_stage_runtime(${target})
endforeach()
add_executable(g01_cpu_ragged_perf_control "${HYLE_SOURCE_DIR}/bench/integration/perf_control_bench.cc")
target_compile_features(g01_cpu_ragged_perf_control PRIVATE cxx_std_23)
target_link_libraries(g01_cpu_ragged_perf_control PRIVATE hyle::compute g01_ragged_nanobench)
set_property(TARGET g01_cpu_ragged_perf_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g01_cpu_ragged_perf_control)
