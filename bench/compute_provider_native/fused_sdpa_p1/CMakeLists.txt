cmake_minimum_required(VERSION 3.28)
project(hyle_native_domain_consumer LANGUAGES CXX)
set(HYLE_MATURE_PEER_ROOT "" CACHE PATH "Frozen source peer checked by the SDK qualification gate")
find_package(Hyle CONFIG REQUIRED COMPONENTS compute)
set(_hyle_root "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_library(nanobench STATIC "${_hyle_root}/bench/nanobench_impl.cc")
target_include_directories(nanobench SYSTEM PUBLIC "${_hyle_root}/third_party/nanobench")
add_executable(hyle_native_domain_bench native_domain_bench.cc)
target_compile_features(hyle_native_domain_bench PRIVATE cxx_std_23)
target_link_libraries(hyle_native_domain_bench PRIVATE hyle::compute nanobench)
set_property(TARGET hyle_native_domain_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime_dependencies(hyle_native_domain_bench)
add_executable(hyle_perf_control_bench "${_hyle_root}/bench/integration/perf_control_bench.cc")
target_compile_features(hyle_perf_control_bench PRIVATE cxx_std_23)
target_link_libraries(hyle_perf_control_bench PRIVATE hyle::compute nanobench)
set_property(TARGET hyle_perf_control_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
