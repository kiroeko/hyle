cmake_minimum_required(VERSION 3.28)
project(hyle_cpu_best_of_consumer LANGUAGES CXX)
find_package(Hyle CONFIG REQUIRED COMPONENTS compute graph)
set(_hyle_root "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_library(nanobench STATIC "${_hyle_root}/bench/nanobench_impl.cc")
target_include_directories(nanobench SYSTEM PUBLIC "${_hyle_root}/third_party/nanobench")
add_executable(hyle_cpu_best_of_bench cpu_best_of_bench.cc)
target_compile_features(hyle_cpu_best_of_bench PRIVATE cxx_std_23)
target_link_libraries(hyle_cpu_best_of_bench PRIVATE hyle::compute hyle::graph nanobench)
set_property(TARGET hyle_cpu_best_of_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime_dependencies(hyle_cpu_best_of_bench)
add_executable(hyle_perf_control_bench "${_hyle_root}/bench/integration/perf_control_bench.cc")
target_compile_features(hyle_perf_control_bench PRIVATE cxx_std_23)
target_link_libraries(hyle_perf_control_bench PRIVATE hyle::compute nanobench)
set_property(TARGET hyle_perf_control_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
