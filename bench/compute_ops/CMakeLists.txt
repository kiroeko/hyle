cmake_minimum_required(VERSION 3.25)
project(g08_topk_p1 LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(HYLE_G08_ROUTE "" CACHE STRING "cpu, cuda, apple or vulkan")
set(HYLE_G08_SOURCE "" CACHE PATH "Frozen current canonical source or verification mirror")
set(HYLE_G08_REFERENCE_DIR "" CACHE PATH "Original85 materialization with source-identity.json")
set(HYLE_G08_REFERENCE_SOURCE "" CACHE PATH "Clean pinned original85 Git checkout")
set(HYLE_G08_VULKAN_PRODUCER "" CACHE PATH "Producer src/compute_route_vulkan directory")
set(HYLE_G08_GLSLANG "" CACHE FILEPATH "Vulkan shader compiler used by the producer")
set(HYLE_G08_GGML_INCLUDE "" CACHE PATH "CPU producer private native headers for original provider only")
if(NOT HYLE_G08_ROUTE MATCHES "^(cpu|cuda|apple|vulkan)$")
  message(FATAL_ERROR "Specify HYLE_G08_ROUTE=cpu, cuda, apple or vulkan")
endif()
if(NOT EXISTS "${HYLE_G08_SOURCE}/bench/nanobench_impl.cc")
  message(FATAL_ERROR "Specify the frozen current HYLE source")
endif()
find_package(Hyle CONFIG REQUIRED COMPONENTS compute)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT Hyle_STATIC)
  message(FATAL_ERROR "The private original provider adapter requires a static SDK")
endif()
set(_original_args "")
if(HYLE_G08_REFERENCE_SOURCE)
  list(APPEND _original_args --original "${HYLE_G08_REFERENCE_SOURCE}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_reference.py"
  --repo "${HYLE_G08_SOURCE}" --reference "${HYLE_G08_REFERENCE_DIR}"
  --route "${HYLE_G08_ROUTE}" --output "${CMAKE_CURRENT_BINARY_DIR}/reference-identity.json" ${_original_args}
  RESULT_VARIABLE _verified OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/reference-identity.log"
  ERROR_FILE "${CMAKE_CURRENT_BINARY_DIR}/reference-identity-error.log")
if(NOT _verified EQUAL 0)
  message(FATAL_ERROR "Original85 source or common private bridge identity failed")
endif()
include("${HYLE_G08_REFERENCE_DIR}/sources.cmake")
set(_cpu 0)
set(_cuda 0)
set(_apple 0)
set(_vulkan 0)
set(_rounds 16)
if(HYLE_G08_ROUTE STREQUAL "cpu")
  set(_cpu 1)
  set(_owner compute_provider_ggml)
  set(_arm generic_arm.cc)
elseif(HYLE_G08_ROUTE STREQUAL "cuda")
  set(_cuda 1)
  set(_rounds 104)
  set(_owner compute_provider_cuda)
  set(_arm generic_arm.cc)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
elseif(HYLE_G08_ROUTE STREQUAL "apple")
  if(NOT APPLE)
    message(FATAL_ERROR "The Apple route requires macOS")
  endif()
  set(_apple 1)
  set(_owner compute_route_apple)
  set(_arm apple_arm.cc)
else()
  set(_vulkan 1)
  set(_owner compute_route_vulkan)
  set(_arm vulkan_arm.cc)
endif()

add_library(g08_original85 STATIC ${G08_ORIGINAL_SOURCES})
target_include_directories(g08_original85 PRIVATE
  "${HYLE_G08_REFERENCE_DIR}/src/${_owner}"
  "${HYLE_G08_REFERENCE_DIR}/src/compute"
  "${HYLE_G08_REFERENCE_DIR}/src/compute_backend_cuda"
  "${HYLE_G08_REFERENCE_DIR}/src/compute_ggml_runtime"
  "${HYLE_G08_REFERENCE_DIR}/src/compute_provider_ggml_vulkan")
target_link_libraries(g08_original85 PUBLIC hyle::compute)
if(_cpu)
  if(NOT EXISTS "${HYLE_G08_GGML_INCLUDE}/ggml.h" OR
     NOT EXISTS "${HYLE_G08_GGML_INCLUDE}/ggml-cpu.h")
    message(FATAL_ERROR "Bind original CPU provider to the producer private ggml headers")
  endif()
  target_include_directories(g08_original85 SYSTEM PRIVATE "${HYLE_G08_GGML_INCLUDE}")
  target_sources(g08_original85 PRIVATE reference_provider.cc)
  target_compile_definitions(g08_original85 PRIVATE ggml_cpu=g08_original_ggml_cpu
    HYLE_G08_CUDA=0 HYLE_GGML_CPU_F32_FLASH=1 HYLE_GGML_CPU_IQ1_EXTENSIONS=1
    HYLE_GGML_CPU_PROVIDER_VERSION="ffa4e8b80930029a35991f94e7c8a93cd67730ab+hyle-resource-iq1-e74fc9bb8964d04f")
  target_link_libraries(g08_original85 PRIVATE hyle::_ggml_ggml)
elseif(_cuda)
  target_sources(g08_original85 PRIVATE reference_provider.cc)
  target_compile_definitions(g08_original85 PRIVATE cuda_native=g08_original_cuda_native HYLE_G08_CUDA=1)
  set_target_properties(g08_original85 PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON
    CUDA_RUNTIME_LIBRARY Shared CUDA_ARCHITECTURES 89-real)
  set_property(GLOBAL PROPERTY JOB_POOLS g08_cuda_compile=1)
  set_property(TARGET g08_original85 PROPERTY JOB_POOL_COMPILE g08_cuda_compile)
  target_compile_options(g08_original85 PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--split-compile=3>")
  if(MSVC)
    target_compile_options(g08_original85 PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/Zc:preprocessor,/utf-8>")
  endif()
  target_link_libraries(g08_original85 PRIVATE CUDA::cudart CUDA::cuda_driver)
elseif(_apple)
  target_compile_definitions(g08_original85 PRIVATE lower_tensor=g08_original_lower_tensor
    lower_mature_indexing=g08_original_lower_mature_indexing
    is_mlx_tensor_custom=g08_original_is_mlx_tensor_custom)
else()
  if(NOT EXISTS "${HYLE_G08_VULKAN_PRODUCER}/vulkan_build.h" OR NOT EXISTS "${HYLE_G08_GLSLANG}")
    message(FATAL_ERROR "Bind Vulkan to the producer config and shader compiler")
  endif()
  # Only owned Vulkan declarations are isolated. The archived full include
  # directory is never used: common declarations come from the same current SDK.
  target_include_directories(g08_original85 BEFORE PRIVATE "${HYLE_G08_REFERENCE_DIR}/public-overlay")
  file(READ "${HYLE_G08_VULKAN_PRODUCER}/vulkan_build.h" _config)
  string(REPLACE "hyle::compute::vulkan" "hyle::compute::g08_original_vulkan" _config "${_config}")
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/original-generated/vulkan_build.h" CONTENT "${_config}" @ONLY)
  set(_shaders "${CMAKE_CURRENT_BINARY_DIR}/original-generated/shaders")
  add_custom_command(OUTPUT "${_shaders}/vulkan_shaders.cc" "${_shaders}/vulkan_shaders.h"
    COMMAND "${Python3_EXECUTABLE}" "${HYLE_G08_REFERENCE_DIR}/src/compute_route_vulkan/build_shaders.py"
      --compiler "${HYLE_G08_GLSLANG}"
      --source "${HYLE_G08_REFERENCE_DIR}/third_party/ggml_vulkan_shaders/shaders"
      --tensor-source "${HYLE_G08_REFERENCE_DIR}/src/compute_route_vulkan/shaders"
      --output "${_shaders}"
    DEPENDS "${HYLE_G08_REFERENCE_DIR}/src/compute_route_vulkan/build_shaders.py"
    VERBATIM)
  target_sources(g08_original85 PRIVATE "${_shaders}/vulkan_shaders.cc")
  target_include_directories(g08_original85 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/original-generated" "${_shaders}")
endif()
target_compile_features(g08_original85 PRIVATE cxx_std_23)
set_property(TARGET g08_original85 PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)

add_library(g08_nanobench STATIC "${HYLE_G08_SOURCE}/bench/nanobench_impl.cc")
target_include_directories(g08_nanobench SYSTEM PUBLIC "${HYLE_G08_SOURCE}/third_party/nanobench")
foreach(_role IN ITEMS candidate reference)
  add_library(g08_${_role}_arm STATIC "${_arm}")
  target_compile_features(g08_${_role}_arm PRIVATE cxx_std_23)
  target_compile_definitions(g08_${_role}_arm PRIVATE HYLE_G08_CUDA=${_cuda})
  target_link_libraries(g08_${_role}_arm PUBLIC hyle::compute)
  if(_role STREQUAL "reference")
    target_compile_definitions(g08_${_role}_arm PRIVATE HYLE_G08_REFERENCE=1)
    target_link_libraries(g08_${_role}_arm PUBLIC g08_original85)
    if(_vulkan)
      target_include_directories(g08_${_role}_arm BEFORE PRIVATE "${HYLE_G08_REFERENCE_DIR}/public-overlay")
    endif()
  else()
    target_compile_definitions(g08_${_role}_arm PRIVATE HYLE_G08_REFERENCE=0)
  endif()
  set_property(TARGET g08_${_role}_arm PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
foreach(_variant IN ITEMS primary placement)
  set(_target g08_topk_${_variant})
  if(_variant STREQUAL "placement")
    add_executable(${_target} placement.cc topk_bench.cc)
    target_compile_definitions(${_target} PRIVATE HYLE_G08_PLACEMENT=1)
    if(MSVC)
      target_link_options(${_target} PRIVATE /INCLUDE:g08_topk_placement_prefix)
    elseif(APPLE)
      target_link_options(${_target} PRIVATE LINKER:-u,_g08_topk_placement_prefix)
    else()
      target_link_options(${_target} PRIVATE LINKER:--undefined=g08_topk_placement_prefix)
    endif()
  else()
    add_executable(${_target} topk_bench.cc)
    target_compile_definitions(${_target} PRIVATE HYLE_G08_PLACEMENT=0)
  endif()
  target_compile_features(${_target} PRIVATE cxx_std_23)
  target_compile_definitions(${_target} PRIVATE HYLE_G08_APPLE=${_apple}
    HYLE_G08_ROUTE_NAME="${HYLE_G08_ROUTE}" HYLE_G08_PAIRED_ROUNDS=${_rounds})
  target_include_directories(${_target} PRIVATE "${HYLE_G08_SOURCE}/bench/common")
  target_link_libraries(${_target} PRIVATE g08_candidate_arm g08_reference_arm g08_nanobench)
  if(MSVC)
    target_compile_definitions(${_target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${_target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  set_property(TARGET ${_target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  hyle_stage_runtime(${_target})
endforeach()
add_executable(g08_topk_control "${HYLE_G08_SOURCE}/bench/integration/perf_control_bench.cc")
target_compile_features(g08_topk_control PRIVATE cxx_std_23)
target_link_libraries(g08_topk_control PRIVATE hyle::compute g08_nanobench)
if(MSVC)
  target_compile_definitions(g08_topk_control PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
set_property(TARGET g08_topk_control PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
hyle_stage_runtime(g08_topk_control)
