cmake_minimum_required(VERSION 3.25)
project(cpu_branch_sdk_consumer LANGUAGES CXX)
find_package(Hyle 0.1.0 EXACT CONFIG REQUIRED COMPONENTS compute graph)
find_package(Threads REQUIRED)
set(HYLE_TEST_REPO "" CACHE PATH "Canonical Windows test consumer sources")
set(HYLE_ORACLE_BUILD "" CACHE PATH "Pinned original ggml/GTest oracle binaries")
set(HYLE_GTEST_SOURCE "" CACHE PATH "Pinned GTest source")
set(HYLE_ORACLE_SOURCE "" CACHE PATH "Pinned original ggml source")
if(WIN32)
  set(_oracle_libraries "${HYLE_ORACLE_BUILD}/bin/ggml.lib" "${HYLE_ORACLE_BUILD}/bin/ggml-cpu.lib" "${HYLE_ORACLE_BUILD}/bin/ggml-base.lib")
  set(_gtest_libraries "${HYLE_ORACLE_BUILD}/lib/gtest_main.lib" "${HYLE_ORACLE_BUILD}/lib/gtest.lib")
else()
  set(_oracle_libraries "${HYLE_ORACLE_BUILD}/bin/libggml.a" "${HYLE_ORACLE_BUILD}/bin/libggml-cpu.a" "${HYLE_ORACLE_BUILD}/bin/libggml-base.a")
  set(_gtest_libraries "${HYLE_ORACLE_BUILD}/lib/libgtest_main.a" "${HYLE_ORACLE_BUILD}/lib/libgtest.a")
endif()
if(TARGET hyle::_ggml_ggml)
  # Static SDK exports its own locked original libraries. Use those for both
  # the product dependency and oracle, so a source-build copy cannot mask SDK assets.
  set(_oracle_libraries hyle::_ggml_ggml hyle::_ggml_ggml-cpu hyle::_ggml_ggml-base)
endif()
foreach(entry IN ITEMS direct graph)
  if(entry STREQUAL "direct")
    set(_source "${HYLE_TEST_REPO}/tests/compute_route_cpu/d5_qwen38_official_test.cc" "${HYLE_TEST_REPO}/tests/compute_route_cpu/d5_qwen_gguf_vocabulary_test.cc")
  else()
    set(_source "${HYLE_TEST_REPO}/tests/graph_route_cpu/d5_qwen38_official_test.cc" "${HYLE_TEST_REPO}/tests/graph_route_cpu/d5_qwen_gguf_vocabulary_test.cc")
  endif()
  add_executable(cpu_branch_${entry} ${_source})
  set_target_properties(cpu_branch_${entry} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${entry}")
  target_compile_features(cpu_branch_${entry} PRIVATE cxx_std_23)
  target_compile_definitions(cpu_branch_${entry} PRIVATE HYLE_TEST_TORCH_CPU=1 HYLE_GGML_CPU_IQ1_EXTENSIONS=1)
  if(entry STREQUAL "graph")
    target_compile_definitions(cpu_branch_${entry} PRIVATE HYLE_TEST_GRAPH_ROUTE_CPU=1)
  endif()
  target_include_directories(cpu_branch_${entry} SYSTEM PRIVATE
    "${HYLE_GTEST_SOURCE}/googletest/include" "${HYLE_ORACLE_SOURCE}/include" "${HYLE_ORACLE_SOURCE}/src")
  # Product calls link installed Hyle. Original ggml/GTest are oracle/test
  # dependencies only; no Hyle product source or source-build library is linked.
  target_link_libraries(cpu_branch_${entry} PRIVATE hyle::compute hyle::graph
    ${_gtest_libraries} ${_oracle_libraries} Threads::Threads ${CMAKE_DL_LIBS})
  hyle_stage_runtime_dependencies(cpu_branch_${entry})
endforeach()
