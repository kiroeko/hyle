cmake_minimum_required(VERSION 3.25)
project(hyle_cuda_u5_consumer LANGUAGES CXX)
find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute graph simd)
if(NOT EXISTS "${U5_GTEST_SOURCE}/googletest/src/gtest-all.cc")
  message(FATAL_ERROR "U5 needs the existing locked GoogleTest source for unchanged test oracles")
endif()
block()
  set(BUILD_SHARED_LIBS OFF)
  set(gtest_force_shared_crt ON)
  set(BUILD_GMOCK OFF)
  set(INSTALL_GTEST OFF)
  add_subdirectory("${U5_GTEST_SOURCE}" fixture-gtest EXCLUDE_FROM_ALL)
endblock()
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(cuda_u5_qualification
  "${_root}/tests/compute_provider_cuda/cuda_provider_test.cc"
  "${_root}/tests/compute_provider_cuda/tensor_test.cc")
target_compile_features(cuda_u5_qualification PRIVATE cxx_std_23)
target_link_libraries(cuda_u5_qualification PRIVATE hyle::compute hyle::simd
  hyle::_factor_compute_provider_cuda GTest::gtest_main)
hyle_stage_runtime_dependencies(cuda_u5_qualification)
add_executable(cuda_u5_ordinary "${_root}/evidence/correctness/cuda-u4-defaults/sdk-consumer/main.cc")
target_compile_features(cuda_u5_ordinary PRIVATE cxx_std_23)
target_link_libraries(cuda_u5_ordinary PRIVATE hyle::compute hyle::graph)
hyle_stage_runtime_dependencies(cuda_u5_ordinary)
