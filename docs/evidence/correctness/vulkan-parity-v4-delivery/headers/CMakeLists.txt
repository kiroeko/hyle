cmake_minimum_required(VERSION 3.25)
project(vulkan_v4_public_headers LANGUAGES CXX)
find_package(Hyle 0.1.0 EXACT REQUIRED COMPONENTS compute graph)
set(_headers compute/vulkan.h compute/detail/vulkan_plan.h graph/vulkan.h)
foreach(_mode IN ITEMS on off)
  set(_sources)
  foreach(_header IN LISTS _headers)
    string(REPLACE "/" "_" _name "${_header}")
    set(_file "${CMAKE_CURRENT_BINARY_DIR}/${_mode}_${_name}.cc")
    set(_prefix "")
    if(_mode STREQUAL "off")
      set(_prefix "#undef HYLE_COMPUTE_VULKAN\n")
    endif()
    file(WRITE "${_file}" "${_prefix}#include <hyle/${_header}>\n")
    list(APPEND _sources "${_file}")
  endforeach()
  add_library(v4_headers_${_mode} OBJECT ${_sources})
  target_compile_features(v4_headers_${_mode} PRIVATE cxx_std_23)
  target_link_libraries(v4_headers_${_mode} PRIVATE hyle::graph)
endforeach()
