1381: ggml_tensor * llama_kv_cache::cpy_v(ggml_context * ctx, ggml_tensor * v_cur, ggml_tensor * v_idxs, int32_t il, const slot_info & sinfo) const {
1382:     GGML_UNUSED(sinfo);
1383: 
1384:     const int32_t ikv = map_layer_ids.at(il);
1385: 
1386:     auto * v = layers[ikv].v;
1387: 
1388:     const int64_t n_embd_head = v_cur->ne[0];
1389:     const int64_t n_head      = v_cur->ne[1];
1390:     const int64_t n_tokens    = v_cur->ne[2];
1391: 
1392:     const int64_t n_embd_gqa = n_embd_head*n_head;
1393: 
1394:     // we can merge dims 0 and 1
1395:     GGML_ASSERT(ggml_row_size(v_cur->type, n_embd_head) == v_cur->nb[1]);
1396: 
1397:     const int64_t n_stream = v->ne[2];
1398: 
1399:     // take this branch when FA is enabled (the V cache is not transposed)
1400:     if (!v_trans) {
1401:         v_cur = ggml_view_2d(ctx, v_cur, n_embd_gqa, n_tokens, v_cur->nb[2], 0);
1402: 
1403:         if (n_stream > 1) {
1404:             const int64_t kv_size = get_size();
1405: 
1406:             assert(n_embd_gqa == v->ne[0]);
1407:             assert(kv_size    == v->ne[1]);
1408: 
1409:             // merge the buffer across all streams because the idxs are global
1410:             v = ggml_reshape_2d(ctx, v, n_embd_gqa, kv_size*n_stream);
1411:         }
1412: 
1413:         return ggml_set_rows(ctx, v, v_cur, v_idxs);
1414:     }
1415: 
1416:     if (ggml_row_size(v_cur->type, n_embd_gqa) == v_cur->nb[2]) {
1417:         // we can merge dims 0, 1 and 2
1418:         v_cur = ggml_reshape_2d(ctx, v_cur, n_embd_gqa, n_tokens);
1419:     } else {
1420:         // otherwise -> make a copy to get contiguous data
1421:         v_cur = ggml_cont_2d   (ctx, v_cur, n_embd_gqa, n_tokens);
1422:     }
1423: 
1424:     // [TAG_V_CACHE_VARIABLE]
1425:     if (n_embd_gqa < v->ne[0]) {
1426:         v_cur = ggml_pad(ctx, v_cur, v->ne[0] - n_embd_gqa, 0, 0, 0);
1427:     }
1428: 
1429:     // in this branch the v_idxs are constructed in such a way that each row is a single head element
1430:     ggml_tensor * v_view = ggml_reshape_2d(ctx, v, 1, ggml_nelements(v));
1431: 
1432:     v_cur = ggml_reshape_2d(ctx, v_cur, 1, ggml_nelements(v_cur));
1433: 
1434:     return ggml_set_rows(ctx, v_view, v_cur, v_idxs);
1435: }
